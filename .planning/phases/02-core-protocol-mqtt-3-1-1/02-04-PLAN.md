    ```java
        Set<Subscription> subs = subscriptionRegistry.getSubscriptions(topicName);
        for (Subscription sub : subs) {
            if (sub.getClientId().equals(clientId)) {
                continue; // don't deliver to publisher itself unless subscribed
                // Actually MQTT spec says publisher CAN receive its own messages if subscribed
                // Remove this check — let publishers receive their own published messages if subscribed
            }
            int deliveryQos = Math.min(publishQos, sub.getQos()); // QoS downgrade
            ClientSessionCtx subscriberCtx = sub.getSessionCtx();
            if (subscriberCtx.getState() != SessionState.CONNECTED) continue;

            if (deliveryQos == 0) {
                // QoS 0: fire and forget
                subscriberCtx.getChannel().writeAndFlush(
                    messageGenerator.createPublish(topicName, 0, payload, false, false, 0));
            } else if (deliveryQos == 1) {
                // QoS 1: assign packet ID, track in outbound map
                int outPktId = subscriberCtx.getPacketIdAllocator().nextPacketId();
                PublishMsg outMsg = PublishMsg.builder().topicName(topicName).qos(1).payload(payload).retain(false).packetId(outPktId).build();
                subscriberCtx.getOutboundQos1().put(outPktId, outMsg);
                subscriberCtx.getChannel().writeAndFlush(
                    messageGenerator.createPublish(topicName, 1, payload, false, false, outPktId));
            } else { // deliveryQos == 2
                int outPktId = subscriberCtx.getPacketIdAllocator().nextPacketId();
                PublishMsg outMsg = PublishMsg.builder().topicName(topicName).qos(2).payload(payload).retain(false).packetId(outPktId).build();
                subscriberCtx.getOutboundQos2().put(outPktId, outMsg);
                subscriberCtx.getChannel().writeAndFlush(
                    messageGenerator.createPublish(topicName, 2, payload, false, false, outPktId));
            }
        }
    }
    ```
    NOTE: The `retain` flag in outgoing messages to subscribers is `false` per MQTT spec — retained flag is only set for retained messages delivered on subscribe (Plan 05).
